\begin{frame}{Pipeline, with the measurements on it}
  \begin{center}
    \includegraphics[width=\linewidth]{fig_pipeline}
  \end{center}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Two processors, one budget}
        \small
        The Hexagon runs the network. The Kryo cores run letterboxing, the
        colour/layout conversion, decoding and NMS. Both are in the frame
        budget; only one is usually reported.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{The NPU is the small term}
        \small
        \textbf{29\%} of the frame goes to the NPU, \textbf{54\%} to decode
        and NMS on the CPU. Making the network faster barely moves the total.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Quantization strategy}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{block}{What AI Hub can actually do}
        \small
        \texttt{submit\_quantize\_job} takes one dtype for weights and one for
        activations, applied to the \emph{whole} model:
        \smallskip
        \begin{tabular}{@{}ll@{}}
          \textbf{0} & FP32 reference \\
          \textbf{1} & W8A8 \\
          \textbf{2} & W8A16 \\
          \textbf{3} & W16A16 \\
          \textbf{4} & W4A16 \\
        \end{tabular}
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{alertblock}{Two rungs we had to drop}
        \small
        \textbf{W8A8 + INT16 head} and \textbf{lite\_mp 10\% INT16} are
        per-layer mixed precision. Qualcomm's documentation states this is
        \emph{``currently not supported''} for quantize jobs, and the API
        exposes no per-op override.
        \smallskip

        Doing it would mean running AIMET locally instead of AI Hub.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \begin{exampleblock}{Protocol}
    \small
    Calibration: 64 tiles from VisDrone-DET-train, seeded, preprocessed by
    importing the runtime's own \texttt{letterbox}
    (\texttt{sha f62e17d4e7cd9946}). Every configuration is profiled on
    QCS8550 \emph{and} checked for whether it still detects --- 4 models
    $\times$ 4 precisions.
  \end{exampleblock}
\end{frame}

\begin{frame}{The fastest precision is the one that is broken}
  \begin{center}
    \includegraphics[width=0.96\linewidth]{fig_quant_trap}
  \end{center}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{W8A8 fails five times out of five}
        \small
        Every class score becomes exactly \textbf{0.0} while box regression
        stays healthy. Three architectures, both export modes --- a property
        of the detect head, not of one model.
      \end{alertblock}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{And W4A16 is worse than broken}
        \small
        It degrades silently: YOLO11n keeps \textbf{29} valid boxes against
        FP32's 244, correlation drops to 0.73 --- \emph{and} it is slower than
        W16A16. More compression bought size, not speed.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{exampleblock}{}
    \centering
    \textbf{8 of 16 quantized configurations are unusable, and every one of
    them reports good latency with zero operator fallback.}
  \end{exampleblock}
\end{frame}

% ------------------------------------------------------------ NEW B ---
\begin{frame}{YOLO26 must not be exported end-to-end}
  \begin{center}
    \includegraphics[width=0.9\linewidth]{fig_e2e_matrix}
  \end{center}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{block}{Why it happens}
        \small
        Removing NMS moves the box decode \emph{inside} the graph: anchor-grid
        offsets spanning 0--640 added to regressed distances. That sum needs
        more dynamic range than a single INT16 scale holds.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{alertblock}{Why no metric catches it}
        \small
        Confidence still tracks FP32 at \textbf{0.9995} and the class labels
        are right. Only the geometry is gone --- every box collapses to one
        point. mAP would report a bad model, not a broken export.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \begin{exampleblock}{The fix, and the cost}
    \small
    Disable \texttt{end2end} before export --- as Qualcomm's own wrapper does
    --- and W8A16 works: 260 detections, 260 valid boxes. The cost is that
    \textbf{NMS returns to the CPU}, which is the 54\% term in the frame budget.
  \end{exampleblock}
\end{frame}

\begin{frame}{Ablation: does the P2 head pay for itself?}
  \begin{center}
    \includegraphics[width=0.94\linewidth]{fig_tradeoff}
  \end{center}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{block}{What the stride-4 branch costs}
        \small
        Anchors go from \textbf{8{,}400 to 34{,}000} at 640. Latency
        \textbf{3.98 $\rightarrow$ 8.53\,ms} (2.1$\times$), for 3.5\% more
        weights. The cost is resolution, not parameters.
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{exampleblock}{What it buys}
        \small
        \textbf{APs $+23.4$\%} over YOLOv8n, against $+10.3$\% on AP overall.
        The margin is largest exactly where the mechanism predicts.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vfill
  \begin{alertblock}{Answer}
    \centering\small
    At \textbf{18\,FPS} the extra 4.5\,ms of NPU time costs about 1.5\,FPS ---
    P2 pays for itself. At a 30\,FPS target it does not.
    \textbf{The frame-rate requirement decides, not the mAP table.}
  \end{alertblock}
\end{frame}

  \begin{center}

  \begin{center}
    \small
    \begin{tabular}{@{}lrrr@{}}
      \toprule
      \textbf{Source} & \textbf{Idle} & \textbf{6-core load} & \textbf{Separation} \\
      \midrule
      \texttt{battery} ($I \times V$) & 506.2 $\pm$ 6.0\,mW & 507.8 $\pm$ 6.1\,mW & 0.3\,$\sigma$ \\
      USB-PD \texttt{ucsi} rail       & 3.000\,A / 5.000\,V & identical & constant \\
      PMIC ADC \texttt{iin\_fb}       & $-15{,}980$\,$\mu$A & $-16{,}028$\,$\mu$A & 0.1\,$\sigma$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \vfill
  \begin{alertblock}{No joule figure appears anywhere in this work}
    \small
    A counter that cannot separate six busy cores from an idle board cannot
    separate two models either. The USB rail reports the negotiated PD
    contract, not draw. \textbf{Thermals do respond} --- 61\,$^\circ$C idle,
    85\,$^\circ$C under load, clocks throttling from 2016 to 556\,MHz at
    $\sim$60\,s --- so temperature is reported as the proxy, labelled as one.
  \end{alertblock}
\end{frame}

\begin{frame}{Discussion and limitations}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Established}
        \small
        \begin{itemize}\itemsep2pt
          \item Tiling: \textbf{+69.8\% APs} for 3.95$\times$ time.
          \item W8A16 is the only precision usable on every model.
          \item W8A8 breaks the detect head, 5 of 5.
          \item End-to-end export cannot be quantized.
          \item NMS is \textbf{54\%} of the frame budget.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Limitations, stated}
        \small
        \begin{itemize}\itemsep2pt
          \item \textbf{NPU numbers are AI Hub's cloud device}, not our board.
                AI Hub ships QAIRT $\geq$2.45; the board runs 2.28, and its
                runtime refuses the binary outright.
          \item Host \texttt{ms/tile} is excluded --- it moved 25\% between
                runs of the same model.
          \item Three of four models hit the epoch cap still improving, so
                these are 30-epoch numbers, not architecture ceilings.
          \item Tracking is training-free (ByteTrack); no MOTA/IDF1 reported.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \begin{block}{}
    \centering\small
    Every table names where it was measured. Where the three sources
    disagree, the disagreement is the result.
  \end{block}
\end{frame}
